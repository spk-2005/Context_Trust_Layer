import hashlib
import json
import logging
import math
import re
from typing import Sequence

import httpx

from app.llm.base import BaseLLMProvider
from app.llm.schemas import (
    ClaimExtractionResult,
    ContradictionResult,
    ExtractedClaim,
)

logger = logging.getLogger(__name__)


def _hash_embedding(text: str, dim: int = 1536) -> list[float]:
    vector = [0.0] * dim
    words = re.findall(r'\w+', text.lower())
    if not words:
        vector[0] = 1.0
        return vector
    for word in words:
        h = int(hashlib.md5(word.encode('utf-8')).hexdigest(), 16)
        vector[h % dim] += 1.0
    norm = math.sqrt(sum(v * v for v in vector))
    if norm > 0:
        vector = [v / norm for v in vector]
    return vector


def _safe_float(value, default: float = 0.0) -> float:
    try:
        return float(value)
    except (TypeError, ValueError):
        return default


def _safe_str(value, default: str = '') -> str:
    return str(value) if value is not None else default


_CLAIM_EXTRACTION_PROMPT = (
    'You are a precise information-extraction engine for a compliance system.\n\n'
    'Given the document section below, extract every distinct, atomic factual claim '
    'as a JSON array under the key "claims".  Each element must have exactly these fields:\n'
    '  - claim_text        : the complete atomic statement (string)\n'
    '  - subject           : the entity or topic the claim is about (string)\n'
    '  - predicate         : the action, property, relation, or condition (string)\n'
    '  - object            : the value, quantity, state, or target entity (string)\n'
    '  - claim_type        : one of "policy", "technical", "financial", "metric", "compliance"\n'
    '  - confidence        : extraction confidence 0.0-1.0 (float)\n'
    '  - evidence_text_span: the exact verbatim substring from the source text (string)\n\n'
    'Rules:\n'
    '- One claim per discrete fact; split compound sentences into separate claims.\n'
    '- Preserve exact numbers, durations, percentages, and dates.\n'
    '- Do NOT infer or add facts not present in the text.\n'
    '- Return ONLY valid JSON - no markdown, no prose.\n\n'
    'Document title : {document_title}\n'
    'Section heading: {section_title}\n'
    'Section text   :\n{section_content}\n\n'
    'Respond with JSON only, e.g.:\n'
    '{{"claims": [{{"claim_text": "...", "subject": "...", "predicate": "...", "object": "...", '
    '"claim_type": "policy", "confidence": 0.95, "evidence_text_span": "..."}}]}}'
)

_CONTRADICTION_PROMPT = (
    'You are a logical-consistency analyser for a compliance knowledge base.\n\n'
    'Determine whether the two claims below contradict each other, then respond with '
    'a single JSON object containing exactly these fields:\n'
    '  - is_contradiction : true / false\n'
    '  - conflict_type    : one of "direct_contradiction", "numerical_discrepancy", '
    '"temporal_outdated", "complementary", "compatible"\n'
    '  - severity         : one of "high", "medium", "low"\n'
    '  - confidence       : float 0.0-1.0\n'
    '  - reasoning        : one concise sentence explaining the decision\n\n'
    'Claim A: {claim_a}\n\n'
    'Claim B: {claim_b}\n\n'
    'Respond with JSON only.'
)


class OllamaProvider(BaseLLMProvider):
    """Real LLM provider backed by a local Ollama server."""

    def __init__(self) -> None:
        from app.core.config import settings
        self.base_url = settings.ollama_base_url.rstrip('/')
        self.model = settings.ollama_model
        self.embedding_model = settings.ollama_embedding_model
        self.timeout = settings.ollama_timeout
        self.embedding_dim = 1536
        logger.info('OllamaProvider initialised — base_url=%s, model=%s', self.base_url, self.model)

    async def _chat(self, prompt: str, model: str = None) -> str:
        url = f'{self.base_url}/api/chat'
        payload = {
            'model': model or self.model,
            'messages': [{'role': 'user', 'content': prompt}],
            'stream': False,
            'options': {'temperature': 0.0},
        }
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(url, json=payload)
            response.raise_for_status()
        data = response.json()
        return data['message']['content'].strip()

    async def extract_claims(self, document_title: str, section_title: str, section_content: str) -> ClaimExtractionResult:
        if not section_content or not section_content.strip():
            return ClaimExtractionResult(claims=[], extraction_model=self.model, extraction_method='ollama_json_extraction')

        prompt = _CLAIM_EXTRACTION_PROMPT.format(
            document_title=document_title,
            section_title=section_title or 'General',
            section_content=section_content,
        )
        raw = ''
        try:
            raw = await self._chat(prompt)
            raw = re.sub(r'<think>.*?</think>', '', raw, flags=re.DOTALL).strip()
            raw = re.sub(r'^  (?:json)?\s*', '', raw)
            raw = re.sub(r'\s*  $', '', raw)
            parsed = json.loads(raw)
            raw_claims = parsed.get('claims', [])
        except (httpx.HTTPStatusError, httpx.RequestError) as exc:
            logger.warning('Ollama request failed during claim extraction: %s', exc)
            return ClaimExtractionResult(claims=[], extraction_model=self.model, extraction_method='ollama_json_extraction')
        except json.JSONDecodeError:
            logger.warning('Ollama returned non-JSON during extraction. Raw: %s', raw[:500])
            return ClaimExtractionResult(claims=[], extraction_model=self.model, extraction_method='ollama_json_extraction')

        claims = []
        for item in raw_claims:
            if not isinstance(item, dict):
                continue
            claim_text = _safe_str(item.get('claim_text', '')).strip()
            if not claim_text:
                continue
            claims.append(ExtractedClaim(
                claim_text=claim_text,
                subject=_safe_str(item.get('subject', 'Unknown'))[:250],
                predicate=_safe_str(item.get('predicate', 'specifies'))[:250],
                object=_safe_str(item.get('object', claim_text)),
                claim_type=_safe_str(item.get('claim_type', 'policy')),
                confidence=_safe_float(item.get('confidence'), 0.90),
                evidence_text_span=_safe_str(item.get('evidence_text_span', claim_text)),
            ))
        logger.info('OllamaProvider extracted %d claims from section "%s"', len(claims), section_title)
        return ClaimExtractionResult(claims=claims, extraction_model=self.model, extraction_method='ollama_json_extraction')

    async def get_embedding(self, text: str) -> list[float]:
        url = f'{self.base_url}/api/embeddings'
        try:
            async with httpx.AsyncClient(timeout=30) as client:
                response = await client.post(url, json={'model': self.embedding_model, 'prompt': text})
            if response.status_code == 200:
                embedding = response.json().get('embedding', [])
                if embedding:
                    return embedding
        except (httpx.RequestError, httpx.HTTPStatusError) as exc:
            logger.debug('Ollama embedding endpoint unavailable (%s); using hash fallback.', exc)
        return _hash_embedding(text, self.embedding_dim)

    async def get_embeddings(self, texts: Sequence[str]) -> list[list[float]]:
        return [await self.get_embedding(t) for t in texts]

    async def evaluate_contradiction(self, claim_a_text: str, claim_b_text: str) -> ContradictionResult:
        prompt = _CONTRADICTION_PROMPT.format(claim_a=claim_a_text, claim_b=claim_b_text)
        raw = ''
        try:
            raw = await self._chat(prompt)
            raw = re.sub(r'<think>.*?</think>', '', raw, flags=re.DOTALL).strip()
            raw = re.sub(r'^  (?:json)?\s*', '', raw)
            raw = re.sub(r'\s*  $', '', raw)
            parsed = json.loads(raw)
            result = ContradictionResult(
                is_contradiction=bool(parsed.get('is_contradiction', False)),
                conflict_type=_safe_str(parsed.get('conflict_type', 'compatible')),
                severity=_safe_str(parsed.get('severity', 'low')),
                confidence=_safe_float(parsed.get('confidence'), 0.85),
                reasoning=_safe_str(parsed.get('reasoning', 'No reasoning provided.')),
            )
            logger.info('OllamaProvider contradiction eval: is_contradiction=%s, type=%s', result.is_contradiction, result.conflict_type)
            return result
        except (httpx.HTTPStatusError, httpx.RequestError) as exc:
            logger.warning('Ollama request failed during contradiction eval: %s', exc)
        except json.JSONDecodeError:
            logger.warning('Ollama returned non-JSON for contradiction eval. Raw: %s', raw[:400])

        return ContradictionResult(
            is_contradiction=False, conflict_type='compatible', severity='low',
            confidence=0.50, reasoning='Ollama evaluation failed; defaulting to no contradiction.',
        )
